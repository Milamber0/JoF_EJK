cmake_minimum_required(VERSION 3.16)
project(language_selection C CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${repo}/codemp/ui/ui_shared.c")
configure_file("${repo}/codemp/versionfile/jofversion.h.in" "jofversion.h" @ONLY)
function(extract start_marker end_marker)
  file(READ "${repo}/codemp/ui/ui_shared.c" source)
  string(FIND "${source}" "${start_marker}" start)
  string(FIND "${source}" "${end_marker}" end)
  if(start LESS 0 OR end LESS start)
    message(FATAL_ERROR "Cannot extract ${start_marker}")
  endif()
  math(EXPR length "${end} - ${start}")
  string(SUBSTRING "${source}" ${start} ${length} section)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "${section}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "")
extract("qboolean PC_String_Parse(" "qboolean PC_Script_Parse(")
extract("qboolean Rect_ContainsPoint(" "int Menu_ItemsMatchingGroup(")
extract("int Item_Multi_CountSettings(" "// Leaving an edit field")
extract("#ifdef UI_BUILD\nstatic const char *UI_LanguageDisplayName" "qboolean ItemParse_cvarFloatList(")
add_executable(language_selection check.cpp
  "${repo}/codemp/botlib/l_script.cpp"
  "${repo}/codemp/botlib/l_precomp.cpp"
  "${repo}/codemp/botlib/l_memory.cpp"
  "${repo}/codemp/qcommon/q_shared.c"
  "${repo}/shared/qcommon/q_string.c")
target_include_directories(language_selection PRIVATE "${repo}/codemp" "${repo}/shared" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(language_selection PRIVATE BOTLIB _CRT_SECURE_NO_WARNINGS)
set_source_files_properties("${repo}/codemp/qcommon/q_shared.c" PROPERTIES LANGUAGE CXX)
if(MSVC)
  target_compile_options(language_selection PRIVATE /Gy /Gw)
  target_link_options(language_selection PRIVATE /OPT:REF)
else()
  target_compile_options(language_selection PRIVATE -ffunction-sections -fdata-sections)
  if(APPLE)
    target_link_options(language_selection PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(language_selection PRIVATE -Wl,--gc-sections)
  endif()
endif()
enable_testing()
add_test(NAME language_selection COMMAND language_selection "${repo}")
