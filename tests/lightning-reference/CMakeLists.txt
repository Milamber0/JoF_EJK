cmake_minimum_required(VERSION 3.16)
project(lightning_reference_check C)

set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/cgame/fx_force.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" contents)
string(FIND "${contents}" "static void FX_ForceLightningReference(" start)
string(FIND "${contents}" "// Shared per-window budget" end)
if(start LESS 0 OR end LESS start)
  message(FATAL_ERROR "Cannot extract the production reference lightning implementation")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${contents}" ${start} ${length} implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "${implementation}")
add_executable(lightning_reference_check check.c)
add_executable(lightning_forward_arc_check arc_check.c)
set(sound_source "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/cgame/cg_main.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${sound_source}")
file(READ "${sound_source}" sound_contents)
string(FIND "${sound_contents}" "// Environment impacts use the stock spark sounds" sound_start)
string(FIND "${sound_contents}" "trap->S_RegisterSound(\"sound/weapons/force/drain.wav\");" sound_end)
if(sound_start LESS 0 OR sound_end LESS sound_start)
  message(FATAL_ERROR "Cannot extract the environment sound registration")
endif()
math(EXPR sound_length "${sound_end}-${sound_start}")
string(SUBSTRING "${sound_contents}" ${sound_start} ${sound_length} sound_registration)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sound_registration.h" "${sound_registration}")
add_executable(lightning_environment_sound_check sound_check.c)
target_include_directories(lightning_environment_sound_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(lightning_reference_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(lightning_forward_arc_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(NOT MSVC)
  target_link_libraries(lightning_reference_check m)
  target_link_libraries(lightning_forward_arc_check m)
endif()
enable_testing()
add_test(NAME lightning_reference_check COMMAND lightning_reference_check "${CMAKE_CURRENT_SOURCE_DIR}/reference.txt")
foreach(pack_mask RANGE 1 7)
  add_test(NAME lightning_reference_pack_${pack_mask}
    COMMAND lightning_reference_check "${CMAKE_CURRENT_SOURCE_DIR}/reference.txt" ${pack_mask})
endforeach()
add_test(NAME lightning_forward_arc_check COMMAND lightning_forward_arc_check)
add_test(NAME lightning_environment_sound_check COMMAND lightning_environment_sound_check)
