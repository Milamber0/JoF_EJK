cmake_minimum_required(VERSION 3.16)
project(map_load_thread CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(source "${repo}/codemp/client/cl_main.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" contents)
string(FIND "${contents}" "void CL_DownloadsComplete( void ) {" first)
string(FIND "${contents}" "void CL_BeginDownloadConfirm( void ) {" last)
if(first LESS 0 OR last LESS first)
  message(FATAL_ERROR "Cannot find download completion implementation")
endif()
math(EXPR length "${last}-${first}")
string(SUBSTRING "${contents}" ${first} ${length} actual)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "${actual}")
add_executable(map_load_thread check.cpp)
target_compile_features(map_load_thread PRIVATE cxx_std_11)
target_include_directories(map_load_thread PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME map_load_thread COMMAND map_load_thread)
